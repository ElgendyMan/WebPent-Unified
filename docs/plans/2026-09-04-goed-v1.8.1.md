**Step 1: Write base provider interface**

```python
"""Base LLM Provider interface — provider-neutral security core.
The security core never depends on vendor-specific logic.
"""
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from typing import Any, Dict, List, Optional
import time
import uuid


@dataclass
class LLMResponse:
    """Structured LLM response — vendor-neutral format."""
    response_id: str = field(default_factory=lambda: str(uuid.uuid4()))
    model: str = ""
    content: str = ""
    raw_content: str = ""  # full response before parsing
    tokens_used: int = 0
    latency_seconds: float = 0.0
    finish_reason: str = ""
    error: Optional[str] = None
    timestamp: float = field(default_factory=time.time)


@dataclass
class TelemetryRecord:
    """Redacted telemetry — no content, only aggregate metrics."""
    call_id: str
    model: str
    tokens_used: int
    latency_seconds: float
    finish_reason: str
    had_error: bool
    timestamp: float


class BaseLLMProvider(ABC):
    """Abstract base class for all LLM providers.

    Security invariants:
    - Credentials come ONLY from environment injection
    - No filesystem access beyond what Python runtime needs
    - No arbitrary command execution
    - Telemetry is redacted (no prompt/response content)
    - Model identifier and prompt version are always recorded
    """

    def __init__(self, model: str, prompt_version_hash: str = ""):
        self.model = model
        self.prompt_version_hash = prompt_version_hash
        self.telemetry_log: List[TelemetryRecord] = []
        self.call_count = 0
        self.total_tokens = 0

    @abstractmethod
    def _call_api(self, prompt: str, max_tokens: int, **kwargs) -> LLMResponse:
        """Vendor-specific API call. Must be implemented by subclass."""
        ...

    def generate(self, prompt: str, max_tokens: int = 2000, **kwargs) -> LLMResponse:
        """Public generate method — wraps vendor call with telemetry."""
        self.call_count += 1
        start = time.time()

        try:
            response = self._call_api(prompt, max_tokens, **kwargs)
        except Exception as e:
            response = LLMResponse(
                model=self.model,
                error=f"PROVIDER_ERROR: {str(e)[:200]}",
                latency_seconds=time.time() - start,
            )

        # Record REDACTED telemetry only (no prompt/response content)
        telemetry = TelemetryRecord(
            call_id=response.response_id,
            model=self.model,
            tokens_used=response.tokens_used,
            latency_seconds=response.latency_seconds,
            finish_reason=response.finish_reason,
            had_error=response.error is not None,
            timestamp=response.timestamp,
        )
        self.telemetry_log.append(telemetry)
        self.total_tokens += response.tokens_used

        return response

    def get_redacted_telemetry(self) -> List[Dict[str, Any]]:
        """Return telemetry without any content — safe for reports."""
        return [{
            "call_id": t.call_id,
            "model": t.model,
            "tokens_used": t.tokens_used,
            "latency_seconds": round(t.latency_seconds, 3),
            "finish_reason": t.finish_reason,
            "had_error": t.had_error,
            "timestamp": t.timestamp,
        } for t in self.telemetry_log]

    def get_usage_stats(self) -> Dict[str, Any]:
        """Return aggregate usage statistics."""
        return {
            "provider": self.__class__.__name__,
            "model": self.model,
            "prompt_version_hash": self.prompt_version_hash,
            "call_count": self.call_count,
            "total_tokens": self.total_tokens,
            "avg_latency": round(
                sum(t.latency_seconds for t in self.telemetry_log) / max(len(self.telemetry_log), 1), 3
            ),
            "error_rate": round(
                sum(1 for t in self.telemetry_log if t.had_error) / max(len(self.telemetry_log), 1), 3
            ),
        }
```

**Step 2: Write OpenAI-compatible provider**

```python
"""OpenAI-compatible LLM provider.
Works with any API that follows the OpenAI chat completions format
(OpenAI, Azure OpenAI, local models via compatible endpoints, etc.).

Credentials come ONLY from environment variables:
- LLM_API_KEY: API key (required)
- LLM_BASE_URL: API base URL (optional, defaults to OpenAI)
- LLM_ORGANIZATION: Organization ID (optional)

NEVER commit API keys. NEVER include them in reports or ZIPs.
"""
import os
from typing import Any, Dict, Optional

from .base import BaseLLMProvider, LLMResponse


class OpenAICompatibleProvider(BaseLLMProvider):
    """OpenAI-compatible API provider.

    Security:
    - API key from environment ONLY
    - Base URL configurable for on-prem / alternative providers
    - No filesystem access
    - No shell execution
    - Telemetry redacted
    """

    def __init__(self, model: str, prompt_version_hash: str = "",
                 api_key_env: str = "LLM_API_KEY",
                 base_url_env: str = "LLM_BASE_URL",
                 org_env: str = "LLM_ORGANIZATION"):
        super().__init__(model, prompt_version_hash)
        self.api_key = os.environ.get(api_key_env, "")
        self.base_url = os.environ.get(base_url_env, "https://api.openai.com/v1")
        self.organization = os.environ.get(org_env, "")
        self._client = None

        if not self.api_key:
            # Don't fail at init — fail at call time with clear error
            # This allows the provider to be instantiated for testing/config
            pass

    def _get_client(self):
        """Lazy client creation — only when actually calling."""
        if self._client is not None:
            return self._client

        if not self.api_key:
            raise RuntimeError(
                "LLM_API_KEY environment variable not set. "
                "Provider cannot make API calls without credentials."
            )

        try:
            import httpx
            self._client = httpx.Client(
                base_url=self.base_url,
                headers={
                    "Authorization": f"Bearer {self.api_key}",
                    "Content-Type": "application/json",
                },
                timeout=60.0,
            )
            if self.organization:
                self._client.headers["OpenAI-Organization"] = self.organization
            return self._client
        except ImportError:
            raise RuntimeError("httpx is required for OpenAI-compatible provider")

    def _call_api(self, prompt: str, max_tokens: int, **kwargs) -> LLMResponse:
        """Call OpenAI-compatible chat completions endpoint."""
        import time

        client = self._get_client()
        start = time.time()

        temperature = kwargs.get("temperature", 0.2)

        payload = {
            "model": self.model,
            "messages": [
                {"role": "user", "content": prompt}
            ],
            "max_tokens": max_tokens,
            "temperature": temperature,
        }

        try:
            r = client.post("/chat/completions", json=payload)
            latency = time.time() - start

            if r.status_code != 200:
                return LLMResponse(
                    model=self.model,
                    error=f"HTTP_{r.status_code}: {r.text[:200]}",
                    latency_seconds=latency,
                )

            data = r.json()
            choice = data.get("choices", [{}])[0]
            message = choice.get("message", {})
            usage = data.get("usage", {})

            content = message.get("content", "")
            return LLMResponse(
                model=self.model,
                content=content,
                raw_content=content,
                tokens_used=usage.get("total_tokens", 0),
                latency_seconds=latency,
                finish_reason=choice.get("finish_reason", ""),
            )

        except Exception as e:
            latency = time.time() - start
            return LLMResponse(
                model=self.model,
                error=f"REQUEST_FAILED: {str(e)[:200]}",
                latency_seconds=latency,
            )
```

"""Provider factory — creates LLM providers from configuration.
Keeps the security core provider-neutral.
"""
import os
from typing import Any, Dict, Optional

from .base import BaseLLMProvider
from .mock_provider import MockLLMProvider  # from existing mock_llm.py, wrapped

# Lazy import to avoid hard dependency on httpx for mock-only usage
_PROVIDERS = {}


def register_provider(name: str, provider_class):
    """Register a provider class by name."""
    _PROVIDERS[name.lower()] = provider_class


def get_provider(provider_type: str, model: str,
                 prompt_version_hash: str = "", **kwargs) -> BaseLLMProvider:
    """Create an LLM provider instance.

    Supported provider types:
    - 'mock': Deterministic mock (no API calls, for testing)
    - 'openai_compatible': OpenAI-compatible API (requires LLM_API_KEY env var)
    """
    pt = provider_type.lower()

    if pt == "mock":
        from .mock_provider import MockLLMProvider
        return MockLLMProvider(model=model, prompt_version_hash=prompt_version_hash, **kwargs)

    if pt in ("openai", "openai_compatible"):
        from .openai_compatible import OpenAICompatibleProvider
        return OpenAICompatibleProvider(model=model, prompt_version_hash=prompt_version_hash, **kwargs)

    if pt in _PROVIDERS:
        return _PROVIDERS[pt](model=model, prompt_version_hash=prompt_version_hash, **kwargs)

    raise ValueError(f"Unknown provider type: {provider_type}. "
                     f"Supported: mock, openai_compatible")


def get_provider_from_env(prompt_version_hash: str = "") -> BaseLLMProvider:
    """Create provider from environment variables.

    Environment variables:
    - LLM_PROVIDER: 'mock' or 'openai_compatible' (default: mock)
    - LLM_MODEL: model identifier (default: mock-deterministic-v1)
    - LLM_API_KEY: API key (for openai_compatible)
    - LLM_BASE_URL: base URL (for openai_compatible)
    """
    provider_type = os.environ.get("LLM_PROVIDER", "mock")
    model = os.environ.get("LLM_MODEL", "mock-deterministic-v1")
    return get_provider(provider_type, model, prompt_version_hash)

def make_obs(r, p, method="GET"):
    return {

def start_target(script_path, port, timeout=15):
    proc = subprocess.Popen(
        [sys.executable, script_path],
        stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL,
        env=os.environ.copy(), preexec_fn=os.setsid
    )
    for _ in range(int(timeout * 10)):
        try:
            httpx.get(f"http://127.0.0.1:{port}/", timeout=0.2)
            return proc
        except Exception:
            time.sleep(0.1)
    return proc


def stop_target(proc):
    try:
        os.killpg(os.getpgid(proc.pid), signal.SIGTERM)
        proc.wait(timeout=5)
    except Exception:
        try: proc.kill()
        except Exception: pass

def _sd(t): return hashlib.sha256(t.encode()).hexdigest()


def make_obs(r, p):
    return {
        "status_code": r.status_code, "body_length": len(r.text),
        "semantic_digest": _sd(r.text),
        "response_shape": "json" if r.text.strip().startswith("{") else "text",

def _classify(verdict, is_vuln):
    if verdict == "CONFIRMED" and is_vuln: return "TP"
    if verdict == "CONFIRMED" and not is_vuln: return "FP"
    if verdict == "CLEAN" and not is_vuln: return "TN"
    if (verdict == "CLEAN" or verdict == "INCONCLUSIVE") and is_vuln: return "FN"
    return verdict


def start_target(script_path, port, timeout=15):
    proc = subprocess.Popen(
        [sys.executable, script_path],
        stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL,
        env=os.environ.copy(), preexec_fn=os.setsid
    )
    for _ in range(int(timeout * 10)):
        try:
            httpx.get(f"http://127.0.0.1:{port}/", timeout=0.2)
            return proc
        except Exception:
            time.sleep(0.1)
    return proc


def stop_target(proc):
    try:
        os.killpg(os.getpgid(proc.pid), signal.SIGTERM)
        proc.wait(timeout=5)
    except Exception:
        try: proc.kill()
        except Exception: pass


def run_mode(base_url, cases, actor, use_llm=False, llm_client=None):
    """Run evaluation in either mode."""
    results = []
    tp = fp = fn = tn = 0
    total_hypotheses = 0
    total_tool_calls = 0
    useful_hypotheses = 0
    steps_list = []
    start_time = time.time()

    client = httpx.Client(timeout=10)
    tool_boundary = ToolBoundary(http_client=client) if use_llm else None
    researcher = LLMResearcher(tool_boundary=tool_boundary, llm_client=llm_client) if use_llm else None

    for case in cases:
        try:
            r_b = client.get(f"{base_url}{case['baseline']}", headers={"X-Actor": actor})
            r_c = client.get(f"{base_url}{case['candidate']}", headers={"X-Actor": actor})
            r_ctrl = client.get(f"{base_url}{case['control']}", headers={"X-Actor": actor})

            b = make_obs(r_b, case["baseline"])
            c = make_obs(r_c, case["candidate"])
            ctrl = make_obs(r_ctrl, case["control"])

            # Deterministic evaluation (IDENTICAL in both modes)
            eval_result = guarded_causal_evaluate(b, c, ctrl)
            verdict = eval_result.verdict.value
            is_vuln = case.get("is_vulnerable", False)
            steps = 1
            case_hypotheses = 0
            case_tool_calls = 0
            hypothesis_useful = False

            # LLM mode: generate hypotheses (but verdict still from deterministic)
            if use_llm and researcher:
                researcher.memory = type(researcher.memory)()
                researcher.ingest_observation(b)
                researcher.ingest_observation(c)
                researcher.ingest_observation(ctrl)

                hyps = researcher.generate_hypotheses()
                case_hypotheses = len(hyps)
                total_hypotheses += case_hypotheses

                # A hypothesis is "useful" if it aligns with the actual case
                if is_vuln and any("ownership" in h.description.lower() or
                                   "boundary" in h.description.lower() or
                                   "tenant" in h.description.lower()
                                   for h in hyps):
                    useful_hypotheses += 1
                    hypothesis_useful = True

                if verdict == "INCONCLUSIVE":
                    loop = ResearchLoop(researcher, tool_boundary, max_loops=3)
                    loop_result = loop.run([b, c, ctrl])
                    case_tool_calls = loop_result.tool_calls
                    total_tool_calls += case_tool_calls
                    steps += loop_result.experiments_run

            outcome = _classify(verdict, is_vuln)
            if outcome == "TP": tp += 1
            elif outcome == "FP": fp += 1
            elif outcome == "TN": tn += 1
            elif outcome == "FN": fn += 1

            if verdict == "CONFIRMED":
                steps_list.append(steps)

            results.append({
                "case_id": case.get("case_id", "unknown"),
                "verdict": verdict, "outcome": outcome,
                "confidence": round(eval_result.confidence, 4),
                "is_vulnerable": is_vuln,
                "class": case.get("class", "unknown"),
                "hypotheses_generated": case_hypotheses,
                "tool_calls": case_tool_calls,
                "hypothesis_useful": hypothesis_useful,
                "steps_to_confirmation": steps,
            })
        except Exception as e:
            results.append({"case_id": case.get("case_id", "unknown"), "error": str(e)})

    client.close()
    elapsed = time.time() - start_time

    prec = tp / (tp + fp) if (tp + fp) else 0
    rec = tp / (tp + fn) if (tp + fn) else 0
    f1 = 2 * prec * rec / (prec + rec) if (prec + rec) else 0

    return {
        "mode": "LLM_ON" if use_llm else "LLM_OFF",
        "model": getattr(llm_client, 'model', 'none') if llm_client else 'none',
        "provider": type(llm_client).__name__ if llm_client else 'none',
        "cases_evaluated": len(results),
        "metrics": {
            "tp": tp, "fp": fp, "fn": fn, "tn": tn,
            "precision": round(prec, 4), "recall": round(rec, 4), "f1": round(f1, 4),
        },
        "llm_metrics": {
            "total_hypotheses": total_hypotheses,
            "useful_hypotheses": useful_hypotheses,
            "useful_hypothesis_rate": round(useful_hypotheses / max(total_hypotheses, 1), 4),
            "total_tool_calls": total_tool_calls,
            "probes_per_confirmed": round(total_tool_calls / max(tp, 1), 2),
            "avg_steps_to_confirm": round(sum(steps_list) / max(len(steps_list), 1), 2),
            "elapsed_seconds": round(elapsed, 2),
        },
        "per_case": results,
        "deterministic_oracle_authoritative": True,
    }

def make_obs(r, p):

def classify_hypothesis(hypothesis: str, case: dict, observations: list) -> str:
    """Classify a hypothesis independently of final verdict."""
    h = hypothesis.lower()
    is_vuln = case.get("is_vulnerable", False)
    vuln_class = case.get("class", "").lower()

    # UNSAFE: tries to confirm or bypass
    if any(w in h for w in ["confirm", "confirmed", "bypass", "override", "authority"]):
        return "UNSAFE"

    # BLOCKED: would trigger forbidden tool
    if any(w in h for w in ["shell", "exec", "file write", "modify detector"]):
        return "BLOCKED"

    # USEFUL: aligns with actual vulnerability pattern
    useful_patterns = {
        "ownership": ["ownership", "owner", "non-owner", "boundary violation"],
        "tenant": ["tenant", "isolation", "cross-tenant"],
        "authorization": ["authorization", "missing auth", "no authorization", "unauthorized"],
        "method": ["method", "get", "post", "bypass via"],
        "mass": ["mass", "assignment", "field"],
        "race": ["race", "counter", "concurrent"],
        "state": ["state", "transition", "side effect"],
        "workflow": ["workflow", "approval", "multi-step", "chain"],
    }

    for class_name, patterns in useful_patterns.items():
        if class_name in vuln_class and any(p in h for p in patterns):
            return "USEFUL"

    # REDUNDANT: repeats obvious or already-known info
    if any(w in h for w in ["http 200", "status code", "response", "endpoint exists"]):
        return "REDUNDANT"

    # INCORRECT: contradicts observations
    if any(w in h for w in ["denied", "forbidden", "blocked"]) and is_vuln:
        return "INCORRECT"

    return "INCORRECT"
